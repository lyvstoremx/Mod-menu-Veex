name: Veex

on:
  workflow_dispatch:
    inputs:
      password:
        description: "Apple ID password"
        required: true
      ipa_url:
        description: "IPA de respaldo (opcional)"
        required: false
        default: ""

jobs:
  build:
    runs-on: macos-15
    timeout-minutes: 60

    steps:
      - uses: actions/checkout@v4

      - name: Compilar dylib
        run: |
          set -euo pipefail
          mkdir -p out
          xcrun -sdk iphoneos clang -arch arm64 -dynamiclib \
            -isysroot "$(xcrun --sdk iphoneos --show-sdk-path)" \
            -miphoneos-version-min=13.0 \
            -framework UIKit -framework Foundation -framework Security \
            -O2 -fobjc-arc \
            -install_name @executable_path/libveexcore.dylib \
            -o out/libveexcore.dylib src/Veex.m
          ls -la out

      - name: Instalar ipatool
        run: |
          set -u
          mkdir -p "$RUNNER_TEMP/ipa"
          cd "$RUNNER_TEMP/ipa"
          BASE=https://github.com/majd/ipatool/releases/download/v2.7.0
          if ! curl -fsSL "$BASE/ipatool-2.7.0-macos-arm64.tar.gz" -o ipatool.tar.gz; then
            curl -fsSL "$BASE/ipatool-2.7.0-macos-amd64.tar.gz" -o ipatool.tar.gz
          fi
          test -s ipatool.tar.gz || { echo "descarga vacia"; exit 1; }
          tar xzf ipatool.tar.gz
          BIN=$(find . -type f ! -name '*.tar.gz' | head -1)
          cp "$BIN" ./ipatool
          chmod +x ./ipatool
          ./ipatool --version
          echo "$RUNNER_TEMP/ipa" >> "$GITHUB_PATH"

      - name: Cache sesion
        uses: actions/cache@v4
        with:
          path: ~/.ipatool
          key: ipatool-auth-v1

      - name: Login
        env:
          APPLE_ID: ${{ secrets.APPLE_ID }}
          APPLE_PW: ${{ inputs.password }}
        run: |
          set -u
          if ipatool auth info >/dev/null 2>&1; then
            echo "sesion ya guardada"
            exit 0
          fi
          rm -f /tmp/fifo login.log
          mkfifo /tmp/fifo
          ipatool auth login -e "$APPLE_ID" -p "$APPLE_PW" < /tmp/fifo > login.log 2>&1 &
          PID=$!
          exec 3> /tmp/fifo
          echo "esperando codigo en code.txt (10 min)"
          CODE=""
          for i in $(seq 1 60); do
            sleep 10
            C=$(curl -s "https://raw.githubusercontent.com/lyvstoremx/Mod-menu-Veex/main/code.txt?t=$i" | tr -d '[:space:]') || true
            echo "poll $i -> '$C'"
            if echo "$C" | grep -Eq '^[0-9]{6}$'; then
              CODE=$C
              break
            fi
            if ! kill -0 $PID 2>/dev/null; then
              echo "ipatool termino antes de tiempo"
              break
            fi
          done
          cat login.log || true
          if [ -z "$CODE" ]; then
            echo "::error::No llego codigo a code.txt"
            kill $PID 2>/dev/null || true
            exit 1
          fi
          echo "codigo: $CODE"
          echo "$CODE" >&3
          wait $PID || true
          echo "--- salida ---"
          cat login.log || true
          if ! ipatool auth info >/dev/null 2>&1; then
            echo "::error::Login fallo"
            exit 1
          fi
          echo "login ok"

      - name: Descargar IPA oficial
        run: |
          set -u
          mkdir -p work
          ipatool download -b com.dts.freefireth -o "$PWD/work/ff.ipa" --non-interactive 2>&1 | tee dl.log || true
          if [ ! -s work/ff.ipa ]; then
            echo "::error::Descarga fallo. Revisa dl.log arriba."
            ls -la work
            exit 1
          fi
          ls -la work

      - name: Verificar cifrado
        run: |
          set -u
          cd work
          unzip -q ff.ipa -d app
          APP=$(ls -d "$PWD"/app/Payload/*.app | head -1)
          echo "app: $APP"
          UF="$APP/Frameworks/UnityFramework.framework/UnityFramework"
          test -f "$UF" || { echo "sin UnityFramework"; exit 1; }
          otool -l "$UF" | grep -A4 LC_ENCRYPTION_INFO | grep cryptid

      - name: Inyectar
        run: |
          set -euo pipefail
          cd "$GITHUB_WORKSPACE"
          git clone -q --depth 1 https://github.com/Tyilo/insert_dylib.git /tmp/idl
          cd /tmp/idl && xcodebuild -quiet -configuration Release >/dev/null 2>&1
          IDL=$(find /tmp/idl -name insert_dylib -type f -perm +111 | head -1)
          APP=$(ls -d "$GITHUB_WORKSPACE"/work/app/Payload/*.app | head -1)
          UF="$APP/Frameworks/UnityFramework.framework/UnityFramework"
          cp "$GITHUB_WORKSPACE/out/libveexcore.dylib" "$APP/Frameworks/UnityFramework.framework/libveexcore.dylib"
          "$IDL" --inplace --strip-codesig --all-yes "@executable_path/libveexcore.dylib" "$UF"
          otool -L "$UF" | grep veexcore
          otool -l "$UF" | grep -A4 LC_ENCRYPTION_INFO | grep cryptid

      - name: Empaquetar
        run: |
          set -euo pipefail
          cd work/app
          zip -qry ../../Veex.ipa Payload
          cd ../..
          ls -la Veex.ipa

      - uses: actions/upload-artifact@v4
        with:
          name: Veex
          path: Veex.ipa
